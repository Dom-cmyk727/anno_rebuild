# MAP-EXTRACT-1 X1: hand-derived vectors of the class-0x3000 terrain helper (AnnoGame 0x10008a80) and of
# the deposit load (AnnoWelt 0x100634e0). Format (one per line, '=>' separates the expectation):
#   terrain NAME w= h= rot= off= table=<2h*2w bytes, row-major> heights=<rows of (W'+1) bytes joined by '/'> => accept|refuse
#   deposit NAME amount=<u16 tons> goods=<id> flag=<0|1> => remaining=<8.8> flag=<0|1>
# Derivations are in docs/sim/salt-extraction.md section 11. Sample points (quarter tiles) per rotation:
#   rot0 (1+2c, 1+2r); rot1 (1+2r, 4w-1-2c); rot2 (4w-1-2c, 4h-1-2r); rot3 (4h-1-2r, 1+2c).
# Interpolated height x4: (1,1): 2a+b+c; (3,3): 2d+b+c; (1,3): b+3c; (3,1): 3b+c  (a b / c d = the tile corners).

# 1x1 footprint, flat 32 everywhere, off 32: every sample is 32, 32-32 = 0.
terrain flat-all-ff w=1 h=1 rot=0 off=32 table=255,255,255,255 heights=32,32/32,32 => accept
# equality refuses: table 0 <= 0
terrain flat-zero-cell w=1 h=1 rot=0 off=32 table=0,255,255,255 heights=32,32/32,32 => refuse
# table 1 > 0 accepts
terrain flat-one-cell w=1 h=1 rot=0 off=32 table=1,1,1,1 heights=32,32/32,32 => accept
# height 255 against table 255 with off 0: 255 <= 255 refuses ("255" is not unconstrained at the top)
terrain top-height w=1 h=1 rot=0 off=0 table=255,255,255,255 heights=255,255/255,255 => refuse
terrain top-height-254 w=1 h=1 rot=0 off=0 table=255,255,255,255 heights=254,254/254,254 => accept
# negative difference passes any byte (signed compare): 32 - 255 = -223 < 0
terrain negative-passes w=1 h=1 rot=0 off=255 table=0,0,0,0 heights=32,32/32,32 => accept
# only the low byte of the offset counts: 0x120 -> 0x20 = 32 -> 0 -> table 0 refuses
terrain offset-low-byte w=1 h=1 rot=0 off=288 table=0,0,0,0 heights=32,32/32,32 => refuse
